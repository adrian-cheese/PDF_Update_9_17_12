# Cifrar archivo con AES
def encrypt_file_with_aes(file_path, aes_key):
    try:
        cipher = AES.new(aes_key, AES.MODE_GCM)
        with open(file_path, 'rb') as f:
            data = f.read()
        ciphertext, tag = cipher.encrypt_and_digest(data)
        encrypted_file_path = f"{file_path}.enc"
        with open(encrypted_file_path, 'wb') as f:
            f.write(cipher.nonce + tag + ciphertext)
        print(f"✅ Archivo cifrado: {file_path}")
        logging.info(f"Archivo cifrado: {file_path}")
        return encrypted_file_path
    except Exception as e:
        print(f"❌ Error al cifrar {file_path}: {e}")
        logging.error(f"Error al cifrar {file_path}: {e}")
        return None

# Cifrar clave AES con RSA
def encrypt_aes_key_with_rsa(aes_key, rsa_public_key):
    try:
        rsa_key = RSA.import_key(rsa_public_key)
        cipher_rsa = PKCS1_OAEP.new(rsa_key)
        encrypted_key = cipher_rsa.encrypt(aes_key)
        return base64.b64encode(encrypted_key).decode('utf-8')
    except Exception as e:
        print(f"❌ Error al cifrar clave AES con RSA: {e}")
        logging.error(f"Error al cifrar clave AES con RSA: {e}")
        traceback.print_exc()
        return None

# Subir archivo a Flask
def upload_file_to_flask(file_path, url=None):
    if url is None:
        url = FLASK_SERVER_URL
        
    try:
        with open(file_path, 'rb') as f:
            files = {'file': (os.path.basename(file_path), f)}
            try:
                response = requests.post(f'{url}/upload', files=files, timeout=15)
                if response.status_code == 200:
                    print(f"✅ Archivo {os.path.basename(file_path)} subido.")
                    logging.info(f"Archivo {os.path.basename(file_path)} subido.")
                    return True
                else:
                    print(f"❌ Error al subir {file_path}: {response.status_code}")
                    logging.error(f"Error al subir {file_path}: {response.status_code}")
                    return False
            except Exception as e:
                print(f"❌ Fallo de conexión al subir {file_path}: {e}")
                logging.error(f"Fallo de conexión al subir {file_path}: {e}")
                return False
    except Exception as e:
        print(f"❌ Error al leer {file_path}: {e}")
        logging.error(f"Error al leer {file_path}: {e}")
        return False
